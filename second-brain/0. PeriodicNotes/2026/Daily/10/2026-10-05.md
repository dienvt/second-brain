SELECT  
      a.id AS account_id,  
      a.name,  
      k.property,  
      JSON_EXTRACT(a.billing_properties, CONCAT('$.', k.property, '.count_from')) AS count_from  
  FROM accounts a  
  JOIN JSON_TABLE(  
      JSON_KEYS(a.billing_properties),  
      '$[*]' COLUMNS (property VARCHAR(100) PATH '$')  
  ) AS k  
  WHERE JSON_TYPE(JSON_EXTRACT(a.billing_properties, CONCAT('$.', k.property))) = 'OBJECT'  
    AND CAST(JSON_EXTRACT(a.billing_properties, CONCAT('$.', k.property, '.count_from')) AS DECIMAL(20,4)) > 1;  


SELECT a.id, a.name
